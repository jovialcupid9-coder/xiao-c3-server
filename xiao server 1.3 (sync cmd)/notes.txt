              void heavyLoad(unsigned long time)
      {
            volatile double x = 0;
            volatile uint32_t y = 1;

            unsigned long start = millis();
            while (millis() - start < time)
            { // run for ~250ms
                  // floating point ops
                  for (int i = 0; i < 1000; i++)
                  {
                        x += sin(i) * cos(i);
                  }
                  // integer ops
                  for (int i = 1; i < 1000; i++)
                  {
                        y *= i;
                        y %= 100000; // prevent overflow
                  }
            }
      }

                                class Json
    {
      // JSON STRUCTURE {"name" : number, "name2 : "String" }
      String &buff;

      Json(String &buff, unsigned int initialSize = 1024) : buff(buff)
      {
        buff.reserve(initialSize);
        buff = "{";
      }

      static bool _add(const String &name, const String &text)
      {
        buff += "\""; // + "
        buff += name;
        buff += ":\"";
        buff += text;
        buff += ",";

        return true; // separator between key values, must be moved at the
                     // beggining with check if it's fisrt cell
      }
      bool add(const String &name, const String &text)
      {
        return _add(name, text);
      }
      bool add(const String &name, const char *str)
      {
        return _add(name, String(text));
      }
      bool add(const String &name, int val) { return _add(name, (String)val); }

      static bool _get(const String &name, const String &output)
      {
        int index = buff.indexOf(name);

        if (index = -1)
          return false;

        output = buff.substring(index, buff.indexOf(","));
        return true;
      }

      bool get(const String &name, const String &output)
      {
        return _get(name, output);
      }
      bool get(const String &name, const int &output)
      {
        String _buff;
        buff.reserve(512);
        _get(name, _buff);

        return _buff.toInt();
      }

      void close() { buff.setCharAt(buff.lastIndexOfbuff(','), '}'); }

      // buff is string so it can be printed directly, it's chunked version
      void print()
      {
        int lastSemicolonIndex = 0;
        int semicolonIndex = 0;

        int lastCommaIndex = 0;
        int commaIndex;

        String name;
        String text;
        name.reserve(256);
        text.reserve(256);

        for (;;)
        {
          semicolonIndex = buff.indexOf(":", lastSemicolonIndex);
          name = buff.substring(1 /*initial open */, semicolonIndex - 1);
          lastSemicolonIndex = semicolonIndex;

          commaIndex = buff.indexOf(",", lastSemicolonIndex);
          if (commaIndex == -1) // last record
          {
            String text =
                buff.substring(lastSemicolonIndex + 1, buff.indexof("}"));
            printer(name, text);
            return;
          }
          else
            String text = buff.substring(lastSemicolonIndex + 1, commaIndex);

          printer(name, text);
        }
      };
